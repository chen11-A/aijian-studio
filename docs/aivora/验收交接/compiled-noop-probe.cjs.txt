const originalResolve = Module._resolveFilename;
Module._resolveFilename = function (...args) {
  const resolved = originalResolve.apply(this, args);
  const real = typeof resolved === "string" && path.isAbsolute(resolved) ? fs.realpathSync(resolved) : resolved;
  if (repoPath(resolved) || repoPath(real)) {
    rejectedRepositoryImports++;
    throw Error("REPOSITORY_RUNTIME_IMPORT");
  }
  return resolved;
};
const sha = (bytes) => crypto.createHash("sha256").update(bytes).digest("hex");
function snapshot(root) {
  const result = [];
  function visit(folder) {
    for (const name of fs.readdirSync(folder).sort()) {
      const file = path.join(folder, name), info = fs.lstatSync(file, { bigint: true });
      assert(!info.isSymbolicLink(), "FIXTURE_LINK_NOT_ALLOWED");
      if (info.isDirectory()) visit(file);
      else result.push({ path: path.relative(root, file), sha: sha(fs.readFileSync(file)), ino: String(info.ino), size: String(info.size), mtime: String(info.mtimeNs), ctime: String(info.ctimeNs) });
    }
  }
  visit(root);
  return result;
}
function denyWrites(counters) {
  const restores = [];
  const replace = (object, name, make) => {
    if (typeof object[name] !== "function") return;
    const previous = object[name];
    object[name] = make(previous);
    restores.push(() => { object[name] = previous; });
  };
  const denied = () => { counters.prepareWrites++; throw Error("PREPARE_MUST_BE_READONLY"); };
  for (const name of ["writeFile", "appendFile", "rename", "rm", "unlink", "mkdir", "rmdir", "copyFile", "cp", "truncate", "chmod", "chown", "utimes", "link", "symlink", "createWriteStream"]) {
    replace(fsp, name, () => denied);
    replace(fs, name, () => denied);
    replace(fs, `${name}Sync`, () => denied);
  }
  const readable = (flags) => typeof flags === "number" ? flags === fs.constants.O_RDONLY : flags === "r" || flags === "rs";
  for (const [object, name] of [[fsp, "open"], [fs, "open"], [fs, "openSync"]])
    replace(object, name, (previous) => function (file, flags, ...args) {
      if (!readable(flags)) return denied();
      return previous.call(this, file, flags, ...args);
    });
  return () => { for (const restore of restores.reverse()) restore(); };
}
